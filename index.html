DATABASE CONFIGURATION ===================================================== */ const DATABASE_CONFIG = { EVENT_NAME: 'World Peace Festival 2026', EVENT_SHORT: 'WPF26', REGISTRATION_FEE: 300, CAPACITY: 600, REGISTRATION_DEADLINE: '2026-09-30T17:00:00+05:30', EVENT_START: '2026-10-08T09:00:00+05:30', EVENT_END: '2026-10-09T22:00:00+05:30', // 24-hour temporary seat hold HOLD_MINUTES: 1440, UPI_ID: 'YOUR-UPI-ID@BANK', PAYMENT_RECEIVER_NAME: 'World Peace Festival', TIMEZONE: 'Asia/Kolkata', // Used later PAYMENT_FOLDER_ID: '' }; /* ===================================================== 2. SHEET DEFINITIONS ===================================================== */ const SHEET_DEFINITIONS = { CONFIG: [ 'Key', 'Value' ], REGISTRATIONS: [ 'Temporary ID', 'Created At', 'Registrant Name', 'Registrant Mobile', 'Participant Count', 'Amount Due', 'Payment Status', 'UTR', 'Screenshot URL', 'Registration Status', 'Hold Created At', 'Hold Expires At', 'Confirmed At', 'Notes' ], PARTICIPANTS: [ 'Permanent ID', 'Temporary ID', 'Participant Number', 'Created At', 'Full Name', 'Age', 'Father/Husband Name', 'Mobile', 'District', 'State', 'Church/Ministry/Organization', 'Designation', 'Recommender', 'Accommodation', 'Food Preference', 'Arrival Date', 'Arrival Time', 'Departure Date', 'Departure Time', 'Registered By', 'Registrant Mobile', 'Status', 'Confirmed At', 'Notes' ], PAYMENTS: [ 'Temporary ID', 'Permanent IDs', 'Expected Amount', 'Submitted Amount', 'Submitted UTR', 'OCR Amount', 'OCR UTR', 'OCR Payee', 'OCR Status', 'OCR Date/Time', 'Screenshot URL', 'Recorded At', 'Payment Status', 'Reconciliation Status', 'Raw OCR Text' ], ACCOMMODATION: [ 'Permanent ID', 'Temporary ID', 'Participant Name', 'Accommodation Required', 'Arrival Date', 'Arrival Time', 'Departure Date', 'Departure Time', 'Room/Bed', 'Status', 'Notes' ], DASHBOARD: [ 'Metric', 'Value', 'Description' ] }; /* ===================================================== 3. GET REQUEST ===================================================== */ function doGet(e) { const action = e && e.parameter && e.parameter.action ? String(e.parameter.action).trim() : ''; /* * Health check * * Example: * ?action=health */ if (action === 'health') { return jsonResponse_({ success: true, service: 'WPF26 API', status: 'OK' }); } /* * Event configuration * * Example: * ?action=config */ if (action === 'config') { return jsonResponse_({ success: true, config: getPublicConfig_() }); } /* * Default response */ return jsonResponse_({ success: true, service: 'WPF26 API', status: 'OK', message: 'World Peace Festival 2026 registration API is active.' }); } /* ===================================================== 4. POST REQUEST ===================================================== */ function doPost(e) { try { const raw = e && e.postData && e.postData.contents ? e.postData.contents : '{}'; const request = JSON.parse(raw); const action = String( request.action || '' ).trim(); /* ----------------------------------------------- CREATE REGISTRATION ----------------------------------------------- */ if (action === 'createRegistration') { const result = handleCreateRegistration_(request); return jsonResponse_({ success: true, registration: result }); } /* ----------------------------------------------- GET REGISTRATION ----------------------------------------------- */ if (action === 'getRegistration') { return jsonResponse_( handleGetRegistration_(request) ); } /* ----------------------------------------------- UNKNOWN ACTION ----------------------------------------------- */ return jsonResponse_({ success: false, error: 'Unknown action: ' + action }); } catch (error) { return jsonResponse_({ success: false, error: error && error.message ? error.message : String(error) }); } } /* ===================================================== 5. JSON RESPONSE ===================================================== */ function jsonResponse_(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } /* ===================================================== 6. PUBLIC CONFIG ===================================================== */ function getPublicConfig_() { return { eventName: getConfigValue_('EVENT_NAME'), eventShort: getConfigValue_('EVENT_SHORT'), registrationFee: Number( getConfigValue_( 'REGISTRATION_FEE' ) ), capacity: Number( getConfigValue_( 'CAPACITY' ) ), registrationDeadline: getConfigValue_( 'REGISTRATION_DEADLINE' ), eventStart: getConfigValue_( 'EVENT_START' ), eventEnd: getConfigValue_( 'EVENT_END' ), holdMinutes: Number( getConfigValue_( 'HOLD_MINUTES' ) ), upiId: getConfigValue_( 'UPI_ID' ), receiverName: getConfigValue_( 'PAYMENT_RECEIVER_NAME' ) }; } /* ===================================================== 7. DATABASE SETUP ===================================================== */ /* * Run this function ONCE manually from Apps Script * before testing the API. */ function setupDatabase() { const ss = SpreadsheetApp .getActiveSpreadsheet(); Object.keys( SHEET_DEFINITIONS ).forEach(function(sheetName) { let sheet = ss.getSheetByName( sheetName ); if (!sheet) { sheet = ss.insertSheet( sheetName ); } const headers = SHEET_DEFINITIONS[ sheetName ]; /* * Add headers if the sheet is empty. */ if ( sheet.getLastRow() === 0 ) { sheet .getRange( 1, 1, 1, headers.length ) .setValues([ headers ]); sheet .getRange( 1, 1, 1, headers.length ) .setFontWeight( 'bold' ); } }); /* * Add missing CONFIG values. * * Existing values are preserved. */ const configSheet = ss.getSheetByName( 'CONFIG' ); const configValues = [ [ 'EVENT_NAME', DATABASE_CONFIG.EVENT_NAME ], [ 'EVENT_SHORT', DATABASE_CONFIG.EVENT_SHORT ], [ 'REGISTRATION_FEE', DATABASE_CONFIG.REGISTRATION_FEE ], [ 'CAPACITY', DATABASE_CONFIG.CAPACITY ], [ 'REGISTRATION_DEADLINE', DATABASE_CONFIG.REGISTRATION_DEADLINE ], [ 'EVENT_START', DATABASE_CONFIG.EVENT_START ], [ 'EVENT_END', DATABASE_CONFIG.EVENT_END ], [ 'HOLD_MINUTES', DATABASE_CONFIG.HOLD_MINUTES ], [ 'UPI_ID', DATABASE_CONFIG.UPI_ID ], [ 'PAYMENT_RECEIVER_NAME', DATABASE_CONFIG.PAYMENT_RECEIVER_NAME ], [ 'PAYMENT_FOLDER_ID', DATABASE_CONFIG.PAYMENT_FOLDER_ID ], [ 'TIMEZONE', DATABASE_CONFIG.TIMEZONE ] ]; const existing = {}; if ( configSheet.getLastRow() > 1 ) { const rows = configSheet .getRange( 2, 1, configSheet.getLastRow() - 1, 2 ) .getValues(); rows.forEach(function(row) { const key = String( row[0] || '' ).trim(); if (key) { existing[key] = true; } }); } configValues.forEach( function(row) { if ( !existing[row[0]] ) { configSheet .appendRow(row); } } ); SpreadsheetApp.flush(); return { success: true, message: 'WPF26 database setup complete.' }; } /* ===================================================== 8. CONFIG HELPER ===================================================== */ function getConfigValue_(key) { const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( 'CONFIG' ); /* * If CONFIG does not exist, * use code defaults. */ if ( !sheet || sheet.getLastRow() < 2 ) { return ( DATABASE_CONFIG[key] || '' ); } const values = sheet .getRange( 2, 1, sheet.getLastRow() - 1, 2 ) .getValues(); for ( let i = 0; i < values.length; i++ ) { const configKey = String( values[i][0] || '' ).trim(); if ( configKey === key ) { return values[i][1]; } } return ( DATABASE_CONFIG[key] || '' ); } /* ===================================================== 9. CREATE REGISTRATION ===================================================== */ function handleCreateRegistration_(request) { request = request || {}; const registrant = request.registrant || {}; const name = String( registrant.name || '' ).trim(); const mobile = String( registrant.mobile || '' ).trim(); const participants = Array.isArray( request.participants ) ? request.participants : []; /* * Basic validation */ if (!name) { throw new Error( 'Registrant name is required.' ); } if (!mobile) { throw new Error( 'Registrant mobile number is required.' ); } if ( participants.length < 1 ) { throw new Error( 'At least one participant is required.' ); } /* * Registration deadline */ const now = new Date(); const deadline = new Date( getConfigValue_( 'REGISTRATION_DEADLINE' ) ); if ( now.getTime() > deadline.getTime() ) { throw new Error( 'Registration deadline has passed.' ); } /* * Validate participants */ participants.forEach( function( participant, index ) { validateParticipant_( participant, index + 1 ); } ); /* * IMPORTANT: * * Lock the whole operation. * * This prevents two people from * taking the same final available * seats simultaneously. */ const lock = LockService .getScriptLock(); if ( !lock.tryLock(10000) ) { throw new Error( 'The registration system is busy. Please try again.' ); } try { /* * Re-check capacity AFTER acquiring * the lock. */ const capacity = Number( getConfigValue_( 'CAPACITY' ) ); const occupied = getOccupiedParticipantCount_(); const requested = participants.length; if ( occupied + requested > capacity ) { throw new Error( 'Not enough registration capacity available.' ); } /* * Generate server-side ID. */ const temporaryId = generateTemporaryId_(); /* * Amount is always calculated * by the server. */ const fee = Number( getConfigValue_( 'REGISTRATION_FEE' ) ); const amountDue = requested * fee; /* * Temporary hold. */ const holdCreatedAt = new Date(); const holdMinutes = Number( getConfigValue_( 'HOLD_MINUTES' ) ); const holdExpiresAt = new Date( holdCreatedAt.getTime() + holdMinutes * 60 * 1000 ); const ss = SpreadsheetApp .getActiveSpreadsheet(); const registrationSheet = getRequiredSheet_( 'REGISTRATIONS' ); const participantSheet = getRequiredSheet_( 'PARTICIPANTS' ); /* * Save registration. */ registrationSheet.appendRow([ temporaryId, holdCreatedAt, name, mobile, requested, amountDue, 'PENDING', '', '', 'ON_HOLD', holdCreatedAt, holdExpiresAt, '', '' ]); /* * Save EVERY participant now. * * This is important. * * We no longer create blank * participant records after payment. */ const participantRows = participants.map( function( participant, index ) { return [ '', temporaryId, index + 1, holdCreatedAt, participant.fullName, Number( participant.age ), participant.fatherHusbandName || '', participant.mobile || '', participant.district || '', participant.state || '', participant.churchOrganization || '', participant.designation || '', participant.recommender || '', participant.accommodation || 'No', participant.foodPreference || 'Veg', participant.arrivalDate || '', participant.arrivalTime || '', participant.departureDate || '', participant.departureTime || '', name, mobile, 'PENDING PAYMENT', '', '' ]; } ); participantSheet .getRange( participantSheet.getLastRow() + 1, 1, participantRows.length, SHEET_DEFINITIONS.PARTICIPANTS.length ) .setValues( participantRows ); SpreadsheetApp.flush(); return { temporaryId: temporaryId, participantCount: requested, amountDue: amountDue, paymentStatus: 'PENDING', registrationStatus: 'ON_HOLD', holdExpiresAt: formatDateTime_( holdExpiresAt ), upiId: getConfigValue_( 'UPI_ID' ), receiverName: getConfigValue_( 'PAYMENT_RECEIVER_NAME' ) }; } finally { lock.releaseLock(); } } /* ===================================================== 10. PARTICIPANT VALIDATION ===================================================== */ function validateParticipant_( participant, number ) { participant = participant || {}; const fullName = String( participant.fullName || '' ).trim(); const age = Number( participant.age ); const mobile = String( participant.mobile || '' ).trim(); const district = String( participant.district || '' ).trim(); const state = String( participant.state || '' ).trim(); if (!fullName) { throw new Error( 'Participant ' + number + ': Full name is required.' ); } if ( !age || age < 18 || age > 120 ) { throw new Error( 'Participant ' + number + ': Age must be between 18 and 120.' ); } if (!mobile) { throw new Error( 'Participant ' + number + ': Mobile number is required.' ); } if (!district) { throw new Error( 'Participant ' + number + ': District is required.' ); } if (!state) { throw new Error( 'Participant ' + number + ': State is required.' ); } } /* ===================================================== 11. TEMPORARY ID GENERATOR ===================================================== */ function generateTemporaryId_() { const properties = PropertiesService .getScriptProperties(); const key = 'NEXT_TEMP_ID_NUMBER'; let nextNumber = Number( properties.getProperty( key ) ); if ( !nextNumber || nextNumber < 1 ) { nextNumber = 1; } const id = getConfigValue_( 'EVENT_SHORT' ) + '-TMP-' + String( nextNumber ).padStart( 5, '0' ); properties.setProperty( key, String( nextNumber + 1 ) ); return id; } /* ===================================================== 12. GET REGISTRATION ===================================================== */ function handleGetRegistration_( request ) { const temporaryId = String( request.temporaryId || '' ) .trim() .toUpperCase(); if (!temporaryId) { return { success: false, error: 'Temporary Registration ID is required.' }; } const registration = getRegistration_( temporaryId ); if (!registration) { return { success: false, error: 'Registration not found.' }; } /* * Automatically expire old holds. */ if ( registration.registrationStatus === 'ON_HOLD' && registration.holdExpiresAt && new Date( registration.holdExpiresAt ).getTime() < new Date().getTime() ) { expireRegistration_( registration.row ); registration.registrationStatus = 'EXPIRED'; registration.paymentStatus = 'EXPIRED'; } const participants = getParticipants_( temporaryId ); return { success: true, registration: { temporaryId: registration.temporaryId, createdAt: formatDateTime_( registration.createdAt ), registrantName: registration.name, registrantMobile: registration.mobile, participantCount: registration.participantCount, amountDue: registration.amountDue, paymentStatus: registration.paymentStatus, registrationStatus: registration.registrationStatus, holdExpiresAt: registration.holdExpiresAt ? formatDateTime_( registration.holdExpiresAt ) : '', confirmedAt: registration.confirmedAt ? formatDateTime_( registration.confirmedAt ) : '', participants: participants } }; } /* ===================================================== 13. FIND REGISTRATION ===================================================== */ function getRegistration_( temporaryId ) { const sheet = getRequiredSheet_( 'REGISTRATIONS' ); if ( sheet.getLastRow() < 2 ) { return null; } const values = sheet .getRange( 2, 1, sheet.getLastRow() - 1, 14 ) .getValues(); const target = String( temporaryId ) .trim() .toUpperCase(); for ( let i = 0; i < values.length; i++ ) { const id = String( values[i][0] || '' ) .trim() .toUpperCase(); if ( id === target ) { return { row: i + 2, temporaryId: values[i][0], createdAt: values[i][1], name: values[i][2], mobile: values[i][3], participantCount: Number( values[i][4] ), amountDue: Number( values[i][5] ), paymentStatus: String( values[i][6] || '' ), utr: values[i][7], screenshotUrl: values[i][8], registrationStatus: String( values[i][9] || '' ), holdCreatedAt: values[i][10], holdExpiresAt: values[i][11], confirmedAt: values[i][12], notes: values[i][13] }; } } return null; } /* ===================================================== 14. GET PARTICIPANTS ===================================================== */ function getParticipants_( temporaryId ) { const sheet = getRequiredSheet_( 'PARTICIPANTS' ); if ( sheet.getLastRow() < 2 ) { return []; } const values = sheet .getRange( 2, 1, sheet.getLastRow() - 1, SHEET_DEFINITIONS.PARTICIPANTS.length ) .getValues(); const target = String( temporaryId ) .trim() .toUpperCase(); const participants = []; values.forEach( function(row) { const rowTempId = String( row[1] || '' ) .trim() .toUpperCase(); if ( rowTempId !== target ) { return; } participants.push({ participantNumber: Number(row[2]), fullName: row[4], age: Number(row[5]), fatherHusbandName: row[6], mobile: row[7], district: row[8], state: row[9], churchOrganization: row[10], designation: row[11], recommender: row[12], accommodation: row[13], foodPreference: row[14], arrivalDate: row[15], arrivalTime: row[16], departureDate: row[17], departureTime: row[18], status: row[21] }); } ); participants.sort( function(a, b) { return ( a.participantNumber - b.participantNumber ); } ); return participants; } /* ===================================================== 15. CAPACITY ===================================================== */ function getOccupiedParticipantCount_() { const sheet = getRequiredSheet_( 'REGISTRATIONS' ); if ( sheet.getLastRow() < 2 ) { return 0; } const values = sheet .getRange( 2, 1, sheet.getLastRow() - 1, 12 ) .getValues(); const now = new Date(); let total = 0; values.forEach( function(row) { const participantCount = Number( row[4] ) || 0; const registrationStatus = String( row[9] || '' ) .trim() .toUpperCase(); /* * Confirmed seats always count. */ if ( registrationStatus === 'CONFIRMED' ) { total += participantCount; return; } /* * ON_HOLD seats count only * until the hold expires. */ if ( registrationStatus === 'ON_HOLD' ) { const expiresAt = row[11] ? new Date( row[11] ) : null; if ( expiresAt && expiresAt.getTime() > now.getTime() ) { total += participantCount; } } } ); return total; } /* ===================================================== 16. EXPIRE REGISTRATION ===================================================== */ function expireRegistration_( rowNumber ) { const sheet = getRequiredSheet_( 'REGISTRATIONS' ); sheet .getRange( rowNumber, 7 ) .setValue( 'EXPIRED' ); sheet .getRange( rowNumber, 10 ) .setValue( 'EXPIRED' ); /* * Also expire participant records. */ const temporaryId = sheet .getRange( rowNumber, 1 ) .getValue(); const participantSheet = getRequiredSheet_( 'PARTICIPANTS' ); if ( participantSheet.getLastRow() < 2 ) { return; } const values = participantSheet .getRange( 2, 1, participantSheet.getLastRow() - 1, 22 ) .getValues(); const target = String( temporaryId ) .trim() .toUpperCase(); values.forEach( function(row, index) { const participantTempId = String( row[1] || '' ) .trim() .toUpperCase(); if ( participantTempId === target ) { participantSheet .getRange( index + 2, 22 ) .setValue( 'EXPIRED' ); } } ); } /* ===================================================== 17. REQUIRED SHEET ===================================================== */ function getRequiredSheet_( sheetName ) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName( sheetName ); if (!sheet) { throw new Error( 'Required sheet "' + sheetName + '" was not found. Run setupDatabase() first.' ); } return sheet; } /* ===================================================== 18. FORMAT DATE/TIME ===================================================== */ function formatDateTime_( date ) { if (!date) { return ''; } return Utilities.formatDate( new Date(date), String( getConfigValue_( 'TIMEZONE' ) || DATABASE_CONFIG.TIMEZONE ), 'dd MMM yyyy, hh:mm a' ); } /* ===================================================== END OF V1 API ===================================================== */
